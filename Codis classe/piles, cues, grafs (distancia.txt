  if (ini == fi) {
    cout << ini << endl;
    return;
  }

  VI pare(n, -1);
  queue<int> Q;
  pare[ini] = -2;
  Q.push(ini);
  while (not Q.empty()) {
    int x = Q.front(); Q.pop();
    for (int y : G[x])
      if (pare[y] == -1) {
        pare[y] = x;
        Q.push(y);

            pila.push(y);

            cout << ' ' << pila.top(); pila.pop();
          }
          cout << endl;
          return;
        }
      }
  }
